```python
def is_palindrome(s: str) -> bool:
    s = s.strip()
    if not s:
        return False
    s = s.replace(" ", "")
    if len(s) <= 1:
        return True
    left, right = 0, len(s) - 1
    while left < right:
        if s[left] != s[right]:
            return False
        left += 1
        right -= 1
    return True
```
